cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(ImmichNativeAI LANGUAGES C CXX)
if(NOT OPENCV_SOURCE)
  message(FATAL_ERROR "Pinned OPENCV_SOURCE is required")
endif()
set(BUILD_LIST core,imgproc,dnn,objdetect CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_WITH_STATIC_CRT ON CACHE BOOL "" FORCE)
set(CPU_BASELINE SSE2 CACHE STRING "" FORCE)
set(CPU_DISPATCH "" CACHE STRING "" FORCE)
foreach(option BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps BUILD_JAVA BUILD_opencv_python2 BUILD_opencv_python3 WITH_IPP WITH_IPP_IW WITH_OPENCL WITH_CUDA WITH_CUDNN WITH_TBB WITH_OPENMP WITH_ITT WITH_EIGEN WITH_LAPACK WITH_QUIRC WITH_FFMPEG WITH_GSTREAMER WITH_MSMF WITH_DSHOW WITH_OPENVINO OPENCV_DNN_OPENCL OPENCV_DNN_CUDA OPENCV_ENABLE_NONFREE OPENCV_GENERATE_PKGCONFIG CV_TRACE PROTOBUF_UPDATE_FILES)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_PROTOBUF ON CACHE BOOL "" FORCE)
set(WITH_PROTOBUF ON CACHE BOOL "" FORCE)
if(NOT PINNED_ZLIB_ROOT)
  message(FATAL_ERROR "Verified zlib 1.3.2 installation is required")
endif()
set(ZLIB_INCLUDE_DIR "${PINNED_ZLIB_ROOT}/include" CACHE PATH "" FORCE)
set(ZLIB_LIBRARY "${PINNED_ZLIB_ROOT}/lib/zs.lib" CACHE FILEPATH "" FORCE)
set(ZLIB_LIBRARY_RELEASE "${PINNED_ZLIB_ROOT}/lib/zs.lib" CACHE FILEPATH "" FORCE)
find_package(ZLIB 1.3.2 EXACT REQUIRED)
set(BUILD_ZLIB OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENCV_SOURCE}" opencv)
add_executable(ai-worker worker.cpp)
set_target_properties(ai-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
target_compile_features(ai-worker PRIVATE cxx_std_17)
target_include_directories(ai-worker PRIVATE "${OPENCV_SOURCE}/modules/core/include" "${OPENCV_SOURCE}/modules/imgproc/include" "${OPENCV_SOURCE}/modules/dnn/include" "${OPENCV_SOURCE}/modules/objdetect/include" "${CMAKE_BINARY_DIR}" "${ZLIB_INCLUDE_DIR}")
target_link_libraries(ai-worker PRIVATE opencv_objdetect opencv_dnn opencv_imgproc opencv_core ZLIB::ZLIB)

# Not shipped. Run actual patched API regressions against the exact worker libraries.
add_executable(native-security-check security-check.cpp)
set_target_properties(native-security-check PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release")
target_compile_features(native-security-check PRIVATE cxx_std_17)
target_include_directories(native-security-check PRIVATE "${OPENCV_SOURCE}/modules/core/include" "${OPENCV_SOURCE}/modules/dnn/include" "${OPENCV_SOURCE}/modules/imgproc/include" "${CMAKE_BINARY_DIR}")
target_link_libraries(native-security-check PRIVATE opencv_dnn opencv_imgproc opencv_core)
